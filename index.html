function showView(id) { document .querySelectorAll('.view') .forEach(view => { view.classList.remove('active'); }); const selectedView = document.getElementById(id); if (selectedView) { selectedView.classList.add('active'); } document .querySelectorAll('.tab-btn') .forEach(btn => { const isActive = btn.dataset.target === id; btn.classList.toggle( 'active', isActive ); if (isActive) { moveIndicator(btn); } }); window.scrollTo({ top: 0, behavior: 'smooth' }); } nav.addEventListener( 'click', event => { const btn = event.target.closest( '.tab-btn' ); if (btn) { showView( btn.dataset.target ); } } );
